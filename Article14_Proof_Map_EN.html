\begin{proposition}[Separation of the fixed feedback channels]
\label{prop:separation}
For the above realisation, varying $(b,c)$ leaves the homogeneous closed-loop state matrix, the plant-input-disturbance response, and the measurement-noise feedback channels unchanged.  It changes the reference-to-output and reference-to-command transfers.
\end{proposition}
\begin{proof}
The state-matrix assertion follows by setting $r=0$ in \eqref{eq:shared-states}--\eqref{eq:shared-command}.  To check the external channels, let $d$ enter the plant input and let the measured output be $y+n$.  Then
\[
 y=P(u+d),\qquad u=C_r^D r-C_y^D(y+n).
\]
Writing $S=(1+PC_y^D)^{-1}$ and solving these two equations gives
\[
 \frac{y}{d}=PS,\quad \frac{u}{d}=-C_y^DPS,
 \qquad \frac{y}{n}=-PC_y^DS,\quad \frac{u}{n}=-C_y^DS.
\]
None contains the reference controller.  In contrast, $y/r=PC_r^DS$ and $u/r=C_r^DS$ contain the adjustable numerator $N_D$.
\end{proof}

Current paper: statement and proof

\begin{theorem}[Exact constrained assignment]
\label{thm:feasibility}
Let $T_i,T_d,T_f>0$, $b\in[\underline b,\overline b]$, and $c\in[\underline c,\overline c]$.
\begin{enumerate}[label=(\alph*)]
\item Under filtered-D placement \eqref{eq:CrD}, $N_D=Q$ has the unique solution
\begin{equation}
 b_D=\frac{\sigma-T_f}{T_i},
 \qquad
 c_D=\frac{\pi-T_f(\sigma-T_f)}{T_iT_d}.
 \label{eq:inverse-D}
\end{equation}
It is feasible if and only if
\begin{align}
 T_f+T_i\underline b&\leq\sigma\leq T_f+T_i\overline b,
 \label{eq:D-feas-1}\\
 T_iT_d\underline c+T_f(\sigma-T_f)&\leq\pi
 \leq T_iT_d\overline c+T_f(\sigma-T_f).
 \label{eq:D-feas-2}
\end{align}
The feasible coefficient set is a parallelogram.
\item Under whole-controller output filtering \eqref{eq:CrO}, $N_O=Q$ has the unique solution
\begin{equation}
 b_O=\frac{\sigma}{T_i},
 \qquad
 c_O=\frac{\pi}{T_iT_d}.
 \label{eq:inverse-O}
\end{equation}
It is feasible if and only if
\begin{equation}
 T_i\underline b\leq\sigma\leq T_i\overline b,
 \qquad
 T_iT_d\underline c\leq\pi\leq T_iT_d\overline c.
 \label{eq:O-feas}
\end{equation}
The feasible coefficient set is a rectangle.
\end{enumerate}
\end{theorem}
\begin{proof}
Matching the coefficients of $s$ and $s^2$ in \eqref{eq:ND} gives
\[
 \begin{bmatrix}\sigma\\ \pi\end{bmatrix}
 =
 \begin{bmatrix}T_f\\0\end{bmatrix}
 +\begin{bmatrix}T_i&0\\T_iT_f&T_iT_d\end{bmatrix}
  \begin{bmatrix}b\\c\end{bmatrix}.
\]
The determinant is $T_i^2T_d>0$.  The first row therefore fixes $b$ uniquely, and substitution in the second fixes $c$ uniquely, giving \eqref{eq:inverse-D}.  Applying the bounds to these two expressions gives \eqref{eq:D-feas-1}--\eqref{eq:D-feas-2}; conversely, those inequalities place the unique inverse inside the weight box.  The image is a parallelogram because an invertible affine map takes each edge of a rectangle to a straight edge and preserves opposite-edge parallelism.  For the output filter the corresponding map is diagonal, with entries $T_i$ and $T_iT_d$, so the inverse and the rectangular region in \eqref{eq:inverse-O}--\eqref{eq:O-feas} follow by the same two coefficient equations.
\end{proof}

\begin{theorem}[Cross-placement command-jump identity]
\label{thm:jump}
Assume a strictly proper plant, zero initial conditions, and a reference step of amplitude $\Delta r$.  Before saturation or rate limiting,
\begin{align}
 \Delta u_D(0^+)&=K_p\left(b_D+\frac{c_DT_d}{T_f}\right)\Delta r,
 \label{eq:jump-D}\\
 \Delta u_O(0^+)&=K_p\frac{c_OT_d}{T_f}\Delta r.
 \label{eq:jump-O}
\end{align}
If the two placements exactly realise the same target \eqref{eq:targetQ}, then
\begin{equation}
 \boxed{\Delta u_D(0^+)=\Delta u_O(0^+)
 =K_p\frac{\pi}{T_iT_f}\Delta r.}
 \label{eq:jump-invariant}
\end{equation}
\end{theorem}
\begin{proof}
For a strictly proper plant, the feedback contribution vanishes in the high-frequency limit.  The initial-value theorem applied to $U(s)$ for $R(s)=\Delta r/s$ gives \eqref{eq:jump-D} and \eqref{eq:jump-O} from \eqref{eq:CrD} and \eqref{eq:CrO}.  Substituting \eqref{eq:inverse-D} gives
\[
 b_D+\frac{c_DT_d}{T_f}
 =\frac{\sigma-T_f}{T_i}
 +\frac{\pi-T_f(\sigma-T_f)}{T_iT_f}
 =\frac{\pi}{T_iT_f}.
\]
Substitution of \eqref{eq:inverse-O} into \eqref{eq:jump-O} gives the same value.
\end{proof}

\begin{corollary}[A necessary damping and frequency bound]
\label{cor:damping-floor}
For an exactly assigned stable complex target under D-branch filtering, $b\geq0$, and the jump limit $\overline u_0>0$,
\begin{equation}
 \sqrt{\frac{K_p|\Delta r|}{\overline u_0T_iT_f}}
 \leq\omega_z\leq\frac{2\zeta_z}{T_f},
 \qquad
 \zeta_z\geq\frac12
 \sqrt{\frac{K_p|\Delta r|T_f}{\overline u_0T_i}}.
 \label{eq:damping-floor}
\end{equation}
When the lower damping bound is below one, it is sharp if only nonnegative weights and the jump limit are imposed.  Upper bounds on the weights can impose further restrictions.
\end{corollary}
\begin{proof}
The jump identity gives the lower bound on $\omega_z$.  The inequality $b_D=(2\zeta_z/\omega_z-T_f)/T_i\geq0$ gives its upper bound.  The interval can be nonempty only if its lower endpoint does not exceed its upper endpoint, which gives the damping bound.  At equality, set $b=0$, choose $\omega_z$ at the common endpoint, and take $c=(T_iT_d\omega_z^2)^{-1}>0$.  This produces exactly the target and saturates the jump limit.  If the resulting damping is below one, the target is complex; if the bound is at least one, no target with $0<\zeta_z<1$ can satisfy these restrictions.
\end{proof}

\end{equation}

the corresponding feedback loop is internally exponentially stable.
\end{proposition}
\begin{proof}
Substituting \eqref{eq:quartic-coefficients} and collecting the $T^2$

\begin{equation}
 \mathcal H_2
 =T^2(v+kC)+vf\{v+f+k(f+d)\}.
 \label{eq:hurwitz-second-expanded}
\end{equation}
Its first summand is strictly positive because $k,C,T>0$, and its
second is nonnegative because $v\geq0$.

$\mathcal H_3=\mathcal H_2(1+k\gamma)-(T^2+vf)^2k/i$
and collect the powers of $v$.  The coefficient of $v^2$ is
$f(1+k\gamma)-kf^2/i=f(1+k)$.
The coefficient of $v$ is
$T^2(1+k\gamma)-2kT^2f/i+

The constant term is
$kT^2C(1+k\gamma)-kT^4/i$.

\begin{align}
 \mathcal H_3={}&f(1+k)v^2
 \nonumber\\
 &+\left[T^2\left\{1+k\left(1-\frac fi\right)\right\}
       +f\{f+k(f+d)\}(1+k\gamma)\right]v
 \nonumber\\
 &+kT^2\left\{C(1+k\gamma)-\frac{T^2}{i}\right\}.
 \label{eq:hurwitz-third-expanded}
\end{align}
The first two terms are nonnegative since $f\leq i$ and $v\geq0$.

\[

\]

\begin{theorem}[Fastest mean response with a fixed delayed feedback loop]
\label{thm:delay-fastest}
Under the preceding hypotheses, the following statements hold.
\begin{enumerate}[label=(\alph*)]
\item Membership of $\mathcal F$ is necessary and sufficient for a
nondecreasing unit-step output from zero to one, subject to $X$.
Every such response has no overshoot and
\begin{equation}
 \boxed{\|G_{b,c}\|_\infty=1.}
 \label{eq:positive-hinf-one}
\end{equation}
\item Its integral absolute tracking error equals its mean response time:
\begin{equation}
 \boxed{m(b):=\int_0^\infty(1-y_{b,c}(t))\,dt
 =\int_0^\infty t g_{b,c}(t)\,dt
 =T_i\left(1-b+\frac{1}{kK_p}\right).}
 \label{eq:delay-mean-area}
\end{equation}
\item The feasible set is convex and compact, even if $X$ is unbounded.
If it is nonempty, the exact global IAE problem is
\begin{equation}
 b_* =\max\{b:(b,c)\in\mathcal F\},\qquad
 m_* =T_i\left(1-b_*+\frac{1}{kK_p}\right).
 \label{eq:delay-linear-program}
\end{equation}
Every feasible $c$ at $b_*$ gives this minimum. The problem has two
variables and a continuum of linear sign constraints; it is a
semi-infinite linear programme.
\end{enumerate}
\end{theorem}
\begin{proof}
The integral action and $P_L(0)=k$ give
$G_{b,c}(0)=1$ for all finite weights. Absolute continuity of the step
response makes nondecreasing output equivalent to $g_{b,c}\geq0$
almost everywhere. Its total mass is one, so positivity gives both
$0\leq y_{b,c}\leq1$ and
\[
 |G_{b,c}(\jj\omega)|
 \leq\int_0^\infty g_{b,c}(t)\,dt=1.
\]
Equality at zero frequency proves \eqref{eq:positive-hinf-one}.
This is the standard DC-gain property of externally positive systems
\citep{GrusslerBurghi2023}; the present argument also covers the exact
delay.

Tonelli's theorem applied to
$1-y(t)=\int_t^\infty g(r)\,dr$ gives the moment identity.
For the value of that moment, use
\begin{equation}
 1-G_{b,c}
 =\frac{P_L^{-1}+C_y-C_r}{P_L^{-1}+C_y}
 =\frac{k^{-1}+(1-b)K_p+O(s)}{K_i/s+O(1)}.
 \label{eq:delay-origin-expansion}
\end{equation}
Therefore $-G_{b,c}'(0)=[k^{-1}+(1-b)K_p]/K_i$, proving
\eqref{eq:delay-mean-area}.

The positive cone in $L^1(0,\infty)$ is closed and convex, so its
inverse image under the affine kernel map is closed and convex.
Moreover, for every nonnegative unit-mass kernel,
\[
 \|b g_b+c g_c\|_1\leq 1+\|g_0\|_1.
\]
The two kernels $g_b,g_c$ are linearly independent. Indeed, a zero
linear combination would imply, after division by the nonzero transfer
$P_LS_L$,
$\beta K_p+\gamma K_ds/(1+T_fs)\equiv0$, which forces
$\beta=\gamma=0$. Norm equivalence on their two-dimensional span
therefore bounds $|b|+|c|$. The full positivity set is compact; its
intersection with closed $X$ is compact as well. Maximising $b$ attains
the minimum because $T_i>0$.
\end{proof}

\begin{equation}

 \label{eq:delay-variance-bound}
\end{equation}
\end{proposition}

\begin{theorem}[MIACE reduction within the fixed-feedback family]
\label{thm:delay-miace}
Assume $\Gamma>0$ in \eqref{eq:delay-plant-moment}. For the fixed
stable loop,
\begin{equation}
 \int_0^\infty e\,dt=m(b),\qquad
 \int_0^\infty v\,dt=\Gamma-m(b).
 \label{eq:delay-signed-areas}
\end{equation}
Let $N_e=\int\max(-e,0)$, $N_v=\int\max(-v,0)$, and $N=N_e+N_v$.
Then
\begin{equation}
 \mathcal M_\alpha^2-\frac{\alpha\Gamma^2}{1+\alpha}
 =\frac{4\alpha}{1+\alpha}N(\Gamma+N)
  +\frac{(A_e-\alpha A_u)^2}{1+\alpha}.
 \label{eq:delay-miace-gap}
\end{equation}
Define the compact convex subset
\begin{equation}
 \mathcal F_+=\{(b,c)\in\mathcal F:v_{b,c}(t)\geq0
                   \text{ for almost every }t\geq0\}.
 \label{eq:delay-positive-command}
\end{equation}
If it is nonempty, write its projection on the $b$ axis as
$[b_-,b_+]$. The exact minimum of MIACE over $\mathcal F_+$ is
obtained at
\begin{align}
 b_\alpha&=1+\frac{1}{kK_p}
                  -\frac{\alpha\Gamma}{(1+\alpha)T_i},\\
 \widehat b_\alpha&=\min\{b_+,\max\{b_-,b_\alpha\}\}.
 \label{eq:delay-miace-projection}
\end{align}
Every feasible $c$ over $\widehat b_\alpha$ attains that constrained
minimum. The universal bound
$\mathcal M_\alpha\geq\Gamma\sqrt{\alpha/(1+\alpha)}$ is attained
within this monotone class if and only if $b_\alpha\in[b_-,b_+]$.
\end{theorem}
\begin{proof}
With $H_L=P_L/k$, the plant equation for the step transforms reads
$Y=H_L(1/s+V)$, where $V$ is the transform of $v$. Hence
$E=(1-H_L)/s-H_LV$. If $P_0$ is unstable, this identity is first
established in a right half-plane and then continued meromorphically
to a neighbourhood of zero. There $H_L$ is analytic and nonzero,
and the error and centred-command transforms have their integrable
limits; clearing the rational plant denominator gives the same
continuation. Taking the limit at zero gives
$\int e+\int v=\Gamma$; \eqref{eq:delay-origin-expansion} gives
the first signed area. The identity
$\int|f|=\int f+2\int\max(-f,0)$ yields
$A_e+A_u=\Gamma+2N$. Insert it in
\[
 A_e^2+\alpha A_u^2
 =\frac{\alpha}{1+\alpha}(A_e+A_u)^2
  +\frac{(A_e-\alpha A_u)^2}{1+\alpha}
\]
to prove \eqref{eq:delay-miace-gap}. The same argument applies to any
zero-initial-state, zero-prehistory settling plant response with
integrable $e,v$, not just this PIDF family.

The step command is affine in the weights. Its nonnegative centred
part therefore adds closed convex constraints, proving the stated
properties of $\mathcal F_+$. On this set the negative areas vanish,
and
\begin{equation}
 \mathcal M_\alpha^2
 =m(b)^2+\alpha[\Gamma-m(b)]^2
 =(1+\alpha)\left[m(b)-\frac{\alpha\Gamma}{1+\alpha}\right]^2
     +\frac{\alpha\Gamma^2}{1+\alpha}.
 \label{eq:delay-miace-quadratic}
\end{equation}
Since $m(b)$ is affine with nonzero slope, minimisation is projection
of its desired value, equivalently projection of $b_\alpha$ onto
$[b_-,b_+]$. Equality in the universal bound gives precisely that
unprojected value.
\end{proof}

\begin{theorem}[A continuum of exact MIACE optima using only $b,c$]
\label{thm:exact-delay-family}
The shared-state loop \eqref{eq:exact-delay-feedback}--
\eqref{eq:exact-delay-weight-path} is internally exponentially stable.
For every $q\in[13/16,1]$, its unit-step output is nondecreasing from
zero to one, and $v=u-1\geq0$. All feedback channels are identical
throughout the family and $\|G_q\|_\infty=1$. The exact areas are
\begin{equation}

 \label{eq:exact-delay-areas}
\end{equation}
For every $\alpha\in[3,5]$, the choice
\begin{equation}

 \label{eq:exact-delay-alpha-path}
\end{equation}
attains the global MIACE lower bound
\begin{equation}
 \mathcal M_\alpha^{\min}
       =\frac94\sqrt{\frac{\alpha}{1+\alpha}}.
 \label{eq:exact-delay-miace-minimum}
\end{equation}
This is global among all zero-initial-state, zero-prehistory settling
responses with integrable error and centred command for the stated
plant, since it attains their common lower bound.

\begin{proof}
Define ordinary nonnegative kernels, zero outside $[0,1/4]$, by
\begin{equation}
 J(t)=E_0e^{-t},\qquad K(t)=E_0e^{-t}-1.
 \label{eq:exact-delay-kernels}
\end{equation}
Integration gives the exact identities
\begin{gather}
 e^{-s/4}=E_0-(1+s)\widehat J(s),\qquad
 R(s):=as+E_0-\widehat J(s)=1+as+s\widehat K(s),\\
 s(as+d)+e^{-s/4}=(1+s)R(s).
 \label{eq:exact-delay-factorisation}
\end{gather}
Let $\gamma=E_0/a$ and $h(t)=e^{-\gamma t}/a$ for $t\geq0$.
The convolution kernel $h*J$ has mass $(E_0-1)/E_0<1$. Hence
\begin{equation}
 r=\mathcal L^{-1}(R^{-1})
   =h*\left(\delta_0+\sum_{n\geq1}(h*J)^{*n}\right)\geq0.
 \label{eq:exact-delay-renewal}
\end{equation}
The same series converges after multiplication by a sufficiently small
positive exponential weight, because $J$ has finite support and the
unweighted inequality is strict. Thus $R^{-1}$ has an exponentially
integrable kernel and no poles in some closed half-plane
$\Re s\geq-\eta$, $\eta>0$. The full unreduced characteristic
factor of the plant and shared PIDF states is proportional to
$(1+s)^3R(s)$. Its other roots are therefore stable. The exponential
stability criterion for finite-dimensional retarded systems proves
internal exponential stability; equivalently the same weighted
convolution bound applies to the forcing generated by an initial
history on its finite delay interval.

The reference and command transfers simplify to
\begin{equation}
 G_q(s)=\frac{e^{-s/4}(1+qs)}{(1+s)^2R(s)},\qquad
 U_q(s)=\frac{1+qs}{R(s)},\qquad
 V_q(s)=\frac{q-a-\widehat K(s)}{R(s)}.
 \label{eq:exact-delay-transfers}
\end{equation}
Here $V_q=(U_q-1)/s$ is the transform of the centred step command.
At $q=1$, $G_1=e^{-s/4}/[(1+s)R]$ has a nonnegative impulse and
unit DC gain. For the whole interval, $G_q/G_1$ is the nonnegative
unit-mass filter in \eqref{eq:exact-delay-prefilter}. Its convolution
with the impulse of $G_1$ proves monotonicity and the unit norm.

It remains to check the command sign, which does not follow from this
output argument. The renewal equation gives
$r'=-\gamma r+(J*r)/a\geq-\gamma r$ and therefore
$r(t-u)\leq e^{\gamma u}r(t)$ for $0\leq u\leq t$.
Writing $\kappa=\int_0^{1/4}e^{\gamma u}K(u)\,du$ yields
\begin{equation}
 v_q=(q-a)r-K*r\geq(q-a-\kappa)r.
 \label{eq:exact-delay-command-certificate}
\end{equation}
For explicit rational bounds, the exponential series and
$n!\geq2^{n-1}$ for $n\geq2$ give
\[
 \frac54<E_0<\frac54+\frac1{28}=\frac97,
 \quad \frac{45}{112}<a<\frac7{16},
 \quad \frac\gamma4<\frac45.
\]
Also $\int K=E_0-5/4<1/28$ and $e<3$, so
$\kappa<3/28$. For $q\geq13/16$,
$q-a>3/8$ and $q-a-\kappa>15/56>0$. This proves $v_q\geq0$
for the complete half-line.

Evaluating $V_q$ at zero gives
$A_u=q-a-\int K=q-7/16$. The plant moment is $\Gamma=2+1/4=9/4$,
so the signed-area balance and the established signs give
\eqref{eq:exact-delay-areas}. Substitution of
\eqref{eq:exact-delay-alpha-path} yields
$A_e=9\alpha/[4(1+\alpha)]$ and $A_u=9/[4(1+\alpha)]$.
Equality in \eqref{eq:delay-miace-gap} proves the global minimum.
As $\alpha$ runs from $3$ to $5$, $q_\alpha$ runs from $1$ to $13/16$.



The fixed feedback loop is internally exponentially stable for every
reference weight, so each zero-initial-state, zero-prehistory tracking
error $e_{b,c}=1-y_{b,c}$ is absolutely integrable. For any
$(b,c)\in[0,1]^2$, the signed-area identity
\eqref{eq:delay-signed-areas}, with $k=1$, gives
\begin{equation}
 \begin{aligned}
 \operatorname{IAE}(b,c)
 &=\int_0^\infty |e_{b,c}(t)|\,dt\\
 &\geq\int_0^\infty e_{b,c}(t)\,dt
   =T_i\left(1-b+\frac1{K_p}\right)\\
 &=(2-\rho)(1-b)+\frac{27}{16}
   \geq\frac{27}{16}.
 \end{aligned}
 \label{eq:exact-delay-box-iae}
\end{equation}
Here $T_i=2-\rho>0$, $T_i/K_p=d=27/16$, and $b\leq1$.
At $q=1$, both weights equal one, and the established monotone response
attains $A_e(1)=27/16$. Thus this endpoint minimises IAE throughout
the entire box.

To compare responses on the path, write
$z(t)=\int_0^t e^{-(t-r)}y_1(r)\,dr$. Since $y_1$ is nonnegative
and nondecreasing, $z(t)\leq y_1(t)$. The identity
$y_q=q y_1+(1-q)z$ proves the asserted pointwise ordering.
\end{proof}
